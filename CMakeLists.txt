cmake_minimum_required(VERSION 3.24)
project(UsbTree VERSION 0.1.0 LANGUAGES C CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build directory")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
include(CTest)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_LIBC ON CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/zip/7f3ae3d57459e59943a4ecfefc8f6277ec6bf540 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/zip/572f249ce1975f98ad9f8aabce512ffa12a52d6c DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL3 imgui)

function(project_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

set(IMGUI_SOURCES ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp)
add_library(imgui STATIC ${IMGUI_SOURCES})
target_include_directories(imgui SYSTEM PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC SDL3::SDL3-static)

add_library(usbtree_app STATIC src/app/model.cpp src/app/access.cpp)
target_include_directories(usbtree_app PUBLIC src)
project_warnings(usbtree_app)
set(PLATFORM_SOURCES src/platform/files.cpp src/platform/usb_stub.cpp)
if(WIN32)
  set(PLATFORM_SOURCES src/platform/files.cpp src/platform/usb_windows.cpp)
elseif(APPLE)
  set(PLATFORM_SOURCES src/platform/files.cpp src/platform/usb_macos.cpp)
endif()
add_library(usbtree_platform STATIC ${PLATFORM_SOURCES} src/platform/notifications.cpp)
target_link_libraries(usbtree_platform PUBLIC usbtree_app SDL3::SDL3-static)
if(WIN32)
  target_link_libraries(usbtree_platform PRIVATE setupapi cfgmgr32)
elseif(APPLE)
  target_link_libraries(usbtree_platform PRIVATE "-framework IOKit" "-framework CoreFoundation")
endif()
project_warnings(usbtree_platform)

set(UI_SOURCES src/ui/theme.cpp src/ui/workspace.cpp src/ui/views.cpp src/ui/renderer.cpp)
add_executable(UsbTree WIN32 src/main.cpp src/platform/desktop.cpp ${UI_SOURCES})
target_link_libraries(UsbTree PRIVATE usbtree_platform imgui)
project_warnings(UsbTree)
set_target_properties(UsbTree PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/portable/UsbTree")
if(WIN32)
  enable_language(RC)
  configure_file(src/platform/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
  target_sources(UsbTree PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app.rc" PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/usbtree.ico")
endif()
if(APPLE)
  target_sources(UsbTree PRIVATE assets/icons/usbtree.icns)
  set_source_files_properties(assets/icons/usbtree.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(UsbTree PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/cmake/Info.plist.in")
endif()
add_custom_target(stage_assets ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/portable/UsbTree/assets"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/README.md" "${CMAKE_BINARY_DIR}/portable/UsbTree/README.md"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/portable/UsbTree/LICENSE"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/docs" "${CMAKE_BINARY_DIR}/portable/UsbTree/docs"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/portable/UsbTree/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL3_SOURCE_DIR}/LICENSE.txt" "${CMAKE_BINARY_DIR}/portable/UsbTree/licenses/SDL.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${imgui_SOURCE_DIR}/LICENSE.txt" "${CMAKE_BINARY_DIR}/portable/UsbTree/licenses/ImGui.txt")
add_dependencies(UsbTree stage_assets)

if(BUILD_TESTING)
  FetchContent_Declare(imgui_test_engine URL https://codeload.github.com/ocornut/imgui_test_engine/zip/7be88e7fa267f251c7271a33d72b1aa5ca04a4cb DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(imgui_test_engine)
  add_library(imgui_tested STATIC ${IMGUI_SOURCES})
  target_include_directories(imgui_tested SYSTEM PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(imgui_tested PUBLIC SDL3::SDL3-static)
  target_compile_definitions(imgui_tested PUBLIC IMGUI_ENABLE_TEST_ENGINE IMGUI_TEST_ENGINE_ENABLE_COROUTINE_STDTHREAD_IMPL=1 IMGUI_TEST_ENGINE_ENABLE_CAPTURE=0)
  file(GLOB ENGINE_SOURCES CONFIGURE_DEPENDS ${imgui_test_engine_SOURCE_DIR}/imgui_test_engine/*.cpp)
  add_library(imgui_engine STATIC ${ENGINE_SOURCES})
  target_include_directories(imgui_engine SYSTEM PUBLIC ${imgui_test_engine_SOURCE_DIR})
  target_link_libraries(imgui_engine PUBLIC imgui_tested)
  add_executable(usbtree_e2e tests/e2e_main.cpp tests/e2e_suite.cpp src/platform/desktop.cpp ${UI_SOURCES})
  target_include_directories(usbtree_e2e PRIVATE tests)
  target_compile_definitions(usbtree_e2e PRIVATE USBTREE_E2E)
  target_link_libraries(usbtree_e2e PRIVATE usbtree_platform imgui_engine)
  project_warnings(usbtree_e2e)
  set_target_properties(usbtree_e2e PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/e2e")
  add_custom_command(TARGET usbtree_e2e POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:usbtree_e2e>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${imgui_test_engine_SOURCE_DIR}/imgui_test_engine/LICENSE.txt" "$<TARGET_FILE_DIR:usbtree_e2e>/ImGui-Test-Engine-LICENSE.txt")
  add_test(NAME imgui_e2e COMMAND usbtree_e2e)
  set_tests_properties(imgui_e2e PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
  add_executable(model_tests tests/model_tests.cpp)
  target_link_libraries(model_tests PRIVATE usbtree_app)
  project_warnings(model_tests)
  add_test(NAME model COMMAND model_tests)
  if(WIN32 OR APPLE)
    add_executable(platform_tests tests/platform_tests.cpp)
    target_link_libraries(platform_tests PRIVATE usbtree_platform)
    project_warnings(platform_tests)
    add_test(NAME native_usb COMMAND platform_tests)
    set_tests_properties(native_usb PROPERTIES TIMEOUT 120)
  endif()
  if(WIN32)
    get_filename_component(MSVC_BIN_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    add_test(NAME portable_folder COMMAND powershell -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_SOURCE_DIR}/scripts/verify-portable.ps1"
      -Executable "$<TARGET_FILE:UsbTree>" -OutputRoot "${CMAKE_BINARY_DIR}/portable-check"
      -Dumpbin "${MSVC_BIN_DIR}/dumpbin.exe")
    set_tests_properties(portable_folder PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  endif()
endif()

install(TARGETS UsbTree RUNTIME DESTINATION . BUNDLE DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES "${SDL3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME SDL.txt)
install(FILES "${imgui_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME ImGui.txt)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME UsbTree)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
set(CPACK_PACKAGE_FILE_NAME "UsbTree-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
